# Phase 0.2 History Sliding Window Implementation Plan

> **For agentic workers:** Use superpowers:executing-plans to implement this plan.

**Goal:** Prevent unbounded `history_text` growth from overflowing agent context windows by truncating at turn boundaries before each `build_prompt()` call.

**Architecture:** Single pure function `truncate_history()` inserted before `build_prompt()` in the WS handler. `history_text` itself keeps growing (full record); only the trimmed copy is sent to agents.

**Tech Stack:** Python 3.11+, regex, pytest

**Spec:** `docs/specs/2026-03-19-phase0.2-history-sliding-window-design.md`

```python
class TestHistorySliding:

    def _h(self, turns):
        """Build a history_text from a list of (agent, text) tuples."""
        return "".join(f"\n[{a}]: {t}\n" for a, t in turns)

    def test_no_truncation_when_under_limit(self):

        h = self._h([("Claude", "hello"), ("Gemini", "world")])
        result = a.truncate_history(h, max_chars=10000)
        assert result == h
        assert a.TRUNCATION_MARKER not in result

    def test_truncation_removes_oldest_turns(self):
        import app as a
        h = self._h([("A", "x" * 100), ("B", "y" * 100), ("C", "z" * 100), ("D", "w" * 100)])
        result = a.truncate_history(h, max_chars=220)
        assert "[D]:" in result
        assert "[A]:" not in result
        assert a.TRUNCATION_MARKER in result

    def test_truncation_marker_appears_once(self):
        import app as a
        h = self._h([("A", "x" * 200), ("B", "y" * 200)])
        result = a.truncate_history(h, max_chars=50)
```

```python
TRUNCATION_MARKER = "[... 較早對話已省略 ...]\n\n"

def truncate_history(history_text: str, max_chars: int) -> str:
    """Sliding window: remove oldest [Agent]: blocks until under max_chars."""
    if len(history_text) <= max_chars:
        return history_text

    segments = re.split(r'(?=\n\[[\w\s\-]+\]: )', history_text)
    while segments and len("".join(segments)) > max_chars:
        segments.pop(0)

    truncated = "".join(segments)
    return TRUNCATION_MARKER + truncated.lstrip("\n")
```
