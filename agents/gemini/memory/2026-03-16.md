with session_base as (
  select 
    user_id, 
    creator_id, 
    range_start(user_session_range) as start_ts,
    range_end(user_session_range) as end_ts,
    duration
  from ${ref('user_session_creator_livestream')}

event_joins as (
  select 
    s.*,
